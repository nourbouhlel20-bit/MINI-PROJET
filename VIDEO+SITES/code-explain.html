pages/contact.js Formulaire de Contact (React + API) 
import { useState } from 'react'

export default function Contact() {
  // États React pour stocker les valeurs
  const [name, setName] = useState('')
  const [message, setMessage] = useState('')
  const [response, setResponse] = useState(null)

  // Appel API au clic sur "Envoyer"
  const submitForm = async (e) => {
    e.preventDefault()
    const res = await fetch('/api/contact', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, message })
    })
    const data = await res.json()
    setResponse(data) // Affiche la réponse
  }

  return (
    <form onSubmit={submitForm}>
      <input value={name}
        onChange={e => setName(e.target.value)} />
      <textarea value={message}
        onChange={e => setMessage(e.target.value)} />
      <button>Envoyer</button>
      {response && <div>{response.reply}</div>}
    </form>
  )
}

💡 Explications

🔄

useState — Hook React qui crée des variables réactives : quand elles changent, l'interface se met à jour automatiquement

🌐

fetch('/api/contact') — Appel HTTP POST vers l'API Next.js intégrée, envoie name + message en JSON

⏳

async/await — Attend la réponse du serveur sans bloquer l'interface (programmation asynchrone)

✅

response && ... — Affiche la réponse API seulement si elle existe (rendu conditionnel React)

pages/api/contact.js API Route Next.js — Côté Serveur 
export default function handler(req, res) {
  // Accepte uniquement les requêtes POST
  if (req.method !== 'POST') {
    return res.status(405)
      .json({ error: 'Method Not Allowed' })
  }

  // Récupère les données envoyées
  const { name, message } = req.body

  // Valide que les champs ne sont pas vides
  if (!name || !message) {
    return res.status(400)
      .json({ error: 'Champs manquants' })
  }

  // Répond avec succès en JSON
  return res.status(200).json({
    success: true,
    reply: `Merci ${name} ! Message reçu.`,
    received: { name, message }
  })
}

💡 Explications

🛡️

Vérification method — Si ce n'est pas un POST, on retourne erreur 405. Bonne pratique de sécurité API

📥

req.body — Contient les données JSON envoyées par le formulaire (name, message)

🔒

Validation — Vérifie que les champs existent avant de traiter (protection contre les données vides)

📤

res.status(200).json() — Retourne une réponse HTTP 200 OK avec un objet JSON

✨

Template literal — `Merci ${name}` insère dynamiquement le nom dans la réponse
